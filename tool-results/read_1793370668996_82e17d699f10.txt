    83→
    84→export default function RegisterProductPage() {
    85→  const router = useRouter();
    86→  const { toast } = useToast();
    87→  const { data: mine } = useMyProducts();
    88→  const { data: provinces } = useProvinces();
    89→  const { data: profile } = usePanelProfile();
    90→  const submit = useSubmitProduct();
    91→
    92→  const [step, setStep] = useState(0);
    93→  const [maxReached, setMaxReached] = useState(0);
    94→  const [draft, setDraft] = useState<Draft>(EMPTY_DRAFT);
    95→  const [done, setDone] = useState(false);
    96→  const [error, setError] = useState('');
    97→  const fileRef = useRef<HTMLInputElement>(null);
    98→  // عکس‌هایی که ثبت محصول موفق بود ولی آپلودشان شکست خورد — برای هشدار/تلاش-دوباره
    99→  const [failedPhotos, setFailedPhotos] = useState<File[]>([]);
   100→  const [retryingPhotos, setRetryingPhotos] = useState(false);
   101→  const [createdId, setCreatedId] = useState<number | null>(null);
   102→
   103→  const { data: cities } = useCities(num(draft.provinceId) || undefined);
   104→
   105→  // ─── پیش‌پر کردن استان/شهر از پروفایل کاربر (خواسته مالک) ───
   106→  // الگوی تنظیم state هنگام رندر (مثل فرم ویرایش پروفایل) — فقط وقتی
   107→  // کاربر خودش چیزی انتخاب نکرده؛ هر وقت خواست عوض می‌کند.
   108→  if (draft.provinceId === '' && draft.cityId === '' && profile?.userInfo.provinceId != null) {
   109→    const pid = profile.userInfo.provinceId;
   110→    const cid = profile.userInfo.cityId;
   111→    setDraft((d) =>
   112→      d.provinceId || d.cityId
   113→        ? d
   114→        : {
   115→            ...d,
   116→            provinceId: String(pid),
   117→            ...(cid != null ? { cityId: String(cid) } : {}),
   118→          },
   119→    );
   120→  }
   121→
   122→  const patch = (p: Partial<Draft>) => setDraft((d) => ({ ...d, ...p }));
   123→
   124→  const goTo = (s: number) => {
   125→    setStep(s);
   126→    setMaxReached((m) => Math.max(m, s));
   127→    setError('');
   128→    if (typeof window !== 'undefined') window.scrollTo({ top: 0, behavior: 'smooth' });
   129→  };
   130→
   131→  // ─── اعتبارسنجی هر مرحله ───
   132→  const stepError = (s: number): string | null => {
   133→    const leaf = draft.leaf;
   134→    if (s === 0 && !leaf) return 'دسته بندی محصول را انتخاب کنید.';
   135→    if (s === 1) {
   136→      if (!draft.productType.trim()) return `نوع ${leaf?.categoryName ?? 'کالا'} خود را وارد کنید.`;
   137→      if (!draft.provinceId) return 'استان مبدا را انتخاب کنید.';
   138→      if (!draft.cityId) return 'شهر مبدا را انتخاب کنید.';
   139→    }
   140→    if (s === 2) {
   141→      if (num(draft.stock) < 1) return 'میزان موجودی محصول را وارد کنید.';
   142→      if (num(draft.minSaleAmount) < 1) return 'حداقل میزان فروش را وارد کنید.';
   143→      if (num(draft.minSaleAmount) > num(draft.stock)) return 'حداقل میزان فروش نمی‌تواند از موجودی بیشتر باشد.';
   144→      if (num(draft.minSalePrice) < 1000) return 'قیمت صحیح را وارد کنید.';
   145→    }
   146→    if (s === 3 && draft.photos.length === 0) return 'حداقل یک تصویر برای محصول بارگذاری کنید.';
   147→    if (s === 4 && draft.description.trim().length < 30) return 'توضیحات محصول را کامل‌تر بنویسید (حداقل ۳۰ حرف).';
   148→    return null;
   149→  };
   150→
   151→  const next = () => {
   152→    const err = stepError(step);
   153→    if (err) {
   154→      setError(err);
   155→      return;
   156→    }
   157→    if (step < 5) goTo(step + 1);
   158→    else void finalize();
   159→  };
   160→
   161→  const finalize = async () => {
   162→    const err = stepError(5);
   163→    if (err) {
   164→      setError(err);
   165→      return;
   166→    }
   167→    try {
   168→      const result = await submit.mutateAsync({
   169→        productName: `${draft.leaf!.categoryName} ${draft.productType.trim()}`,
   170→        subCategoryId: draft.leaf!.id,
   171→        provinceId: num(draft.provinceId),
   172→        cityId: num(draft.cityId),
   173→        stock: num(draft.stock),
   174→        minSaleAmount: num(draft.minSaleAmount),
   175→        minSalePrice: num(draft.minSalePrice),
   176→        description: draft.description,
   177→        features: draft.features.filter((f) => f.key && f.value.trim()),
   178→        photos: draft.photos.map((p) => p.file),
   179→      });
   180→      setFailedPhotos(result.failedPhotos);
   181→      setCreatedId(result.id);
   182→      setDone(true);
   183→    } catch (err) {
   184→      toast({ description: err instanceof Error && err.message ? err.message : 'خطا در ثبت محصول — دوباره تلاش کنید.' });
   185→    }
   186→  };
   187→
   188→  // ─── تلاش دوباره عکس‌های ناموفق — محصول ثبت شده، فقط عکس‌ها مانده ───
   189→  const retryPhotos = async () => {
   190→    if (createdId == null || failedPhotos.length === 0) return;
   191→    setRetryingPhotos(true);
   192→    try {
   193→      const { failed } = await retryProductPhotos(createdId, failedPhotos);
   194→      setFailedPhotos(failed);
   195→      if (failed.length === 0) {
   196→        toast({ description: 'تصاویر محصول با موفقیت ذخیره شدند.' });
   197→      } else {
   198→        toast({ description: 'باز هم ذخیره نشد — دوباره تلاش کنید یا بعداً از ویرایش آگهی عکس بگذارید.' });
   199→      }
   200→    } finally {
   201→      setRetryingPhotos(false);
   202→    }
   203→  };
   204→
   205→  // ─── تصاویر ───
   206→  const addFiles = (files: FileList | null) => {
   207→    if (!files) return;
   208→    let rejected = '';
   209→    const slots = 4 - draft.photos.length;
   210→    Array.from(files)
   211→      .slice(0, Math.max(0, slots))
   212→      .forEach((file) => {
   213→        if (!/image\/jpe?g/.test(file.type)) {
   214→          rejected = 'تصویر از نوع jpeg یا jpg باشد.';
   215→          return;
   216→        }
   217→        if (file.size > 15 * 1024 * 1024) {
   218→          rejected = 'حجم تصویر بسیار بالا است (بیش از ۱۵ مگابایت).';
   219→          return;
   220→        }
   221→      });
   222→    if (rejected) {
   223→      setError(rejected);
   224→      return;
   225→    }
   226→
   227→    // بهینه‌سازی سمت کلاینت (عرض ≤۱۲۰۰، حجم ≤۵۰۰KB) بعد پیش‌نمایش محلی
   228→    void (async () => {
   229→      const loaded: Draft['photos'] = [];
   230→      for (const file of Array.from(files).slice(0, Math.max(0, slots))) {
   231→        try {
   232→          const optimized = await optimizeProductImage(file);
   233→          const url = await new Promise<string>((resolve) => {
   234→            const reader = new FileReader();
   235→            reader.onload = () => resolve(String(reader.result));
   236→            reader.readAsDataURL(optimized);
   237→          });
   238→          loaded.push({ id: Date.now() + Math.random(), url, file: optimized });
   239→        } catch {
   240→          rejected = 'خواندن یکی از تصاویر ممکن نشد.';
   241→        }
   242→      }
   243→      if (loaded.length > 0) {
   244→        setDraft((d) => ({ ...d, photos: [...d.photos, ...loaded].slice(0, 4) }));
   245→      }
   246→      if (rejected) setError(rejected);
   247→    })();
   248→  };
   249→
   250→  const featureOptions = (currentKey: string) =>
   251→    PRODUCT_FEATURE_KEYS.filter((k) => k === currentKey || !draft.features.some((f) => f.key === k));
   252→
   253→  const descLen = draft.description.trim().length;
   254→
   255→  // ─── ظرفیت تکمیل — مودال «امکان ثبت محصول جدید را ندارید!» ───
   256→  // هماهنگ با بک‌اند: فقط pending + confirmed ظرفیت می‌گیرند (رد شده و حذف
   257→  // شده نه — باگ ۲۸-۱۰: فرایت رد شده‌ها را هم می‌شمرد و زودتر بلاک می‌کرد)
   258→  const capacityFull = Boolean(
   259→    mine && mine.items.filter((p) => p.status === 'pending' || p.status === 'confirmed').length >= mine.capacity,
   260→  );
   261→
   262→  if (capacityFull && !done) {
   263→    return (
   264→      <div className="mx-auto max-w-xl">
   265→        <PanelCard className="text-center">
   266→          <span className="mx-auto grid size-16 place-items-center rounded-full bg-red-50 text-red-500">
   267→            <TrendingUp size={28} className="rotate-180" />
   268→          </span>
   269→          <h1 className="mt-4 text-base font-extrabold">امکان ثبت محصول جدید را ندارید!</h1>
   270→          <p className="mt-2 text-sm leading-6 text-muted-foreground">
   271→            برای ثبت محصولات بیشتر، ظرفیت ثبت محصول حساب خود را افزایش دهید.
   272→          </p>
   273→          <Button asChild className="mt-5 rounded-md">
   274→            <Link href="/panel/pricing">افزایش ظرفیت</Link>
   275→          </Button>
   276→        </PanelCard>
   277→      </div>
   278→    );
   279→  }
   280→
   281→  // ─── صفحه موفقیت ───
   282→  if (done) {
   283→    return (
   284→      <div className="mx-auto max-w-xl">
   285→        <PanelCard className="text-center">
   286→          <span className="mx-auto grid size-16 place-items-center rounded-full bg-emerald-50 text-emerald-600">
   287→            <PartyPopper size={28} />
   288→          </span>
   289→          <h1 className="mt-4 text-base font-extrabold">محصول شما با موفقیت ثبت شد</h1>
   290→          <p className="mt-2 text-sm leading-6 text-muted-foreground">
   291→            آگهی «{draft.leaf?.categoryName} {draft.productType}» ثبت شد و پس از بررسی کارشناسان ایمچ، در «محصولات من»
   292→            و نتایج بازار منتشر می‌شود.
   293→          </p>
   294→
   295→          {/* هشدار عکس‌های ذخیره‌نشده — دیگر بی‌صدا گم نمی‌شوند */}
   296→          {failedPhotos.length > 0 && (
   297→            <div className="mt-4 rounded-md border border-amber-300 bg-amber-50 p-3 text-right">
   298→              <p className="text-sm font-bold text-amber-800">
   299→                ذخیره {failedPhotos.length.toLocaleString('fa-IR')} تصویر از {draft.photos.length.toLocaleString('fa-IR')} تصویر ناموفق بود
   300→              </p>
   301→              <p className="mt-1 text-xs leading-5 text-amber-700">
   302→                آگهی ثبت شد اما تصاویر ذخیره نشدند — اتصال اینترنت را بررسی کنید و دوباره تلاش کنید.
   303→              </p>
   304→              <Button
   305→                size="sm"
   306→                className="mt-2 rounded-md"
   307→                disabled={retryingPhotos}
   308→                onClick={() => void retryPhotos()}
   309→              >
   310→                {retryingPhotos ? 'در حال تلاش…' : 'تلاش دوباره ذخیره تصاویر'}
   311→              </Button>
   312→            </div>
   313→          )}
   314→
   315→          <div className="mt-5 flex flex-wrap justify-center gap-2">
   316→            <Button
   317→              variant="outline"
   318→              className="rounded-md"
   319→              onClick={() => {
   320→                setDraft(EMPTY_DRAFT);
   321→                setStep(0);
   322→                setMaxReached(0);
   323→                setDone(false);
   324→                setFailedPhotos([]);
   325→                setCreatedId(null);
   326→              }}
   327→            >
   328→              <RotateCcw size={15} /> ثبت محصول دیگر
   329→            </Button>
   330→            <Button className="rounded-md" onClick={() => router.push('/panel/products?tab=pending')}>
   331→              مشاهده محصول در لیست محصولات
   332→            </Button>
   333→          </div>
   334→        </PanelCard>
   335→      </div>
   336→    );
   337→  }
   338→
   339→  const leaf = draft.leaf;
   340→
   341→  return (
   342→    <div className="mx-auto max-w-3xl space-y-4">
   343→      <h1 className="hidden text-lg font-extrabold lg:block">ثبت محصول</h1>
   344→      <ProductStepper current={step} maxReached={maxReached} onJump={goTo} />
   345→
   346→      {/* ─── مرحله ۱: انتخاب دسته بندی ─── */}
   347→      {step === 0 && (
   348→        <PanelCard>
   349→          <CategoryCascade value={{ path: draft.path, leaf: draft.leaf }} onChange={(v) => patch({ path: v.path, leaf: v.leaf })} />
   350→        </PanelCard>
   351→      )}
   352→
   353→      {/* ─── مرحله ۲: اطلاعات محصول ─── */}
   354→      {step === 1 && (
   355→        <PanelCard className="space-y-5">
   356→          <div className="space-y-2">
   357→            <Label htmlFor="product-type">
   358→              نوع <span className="text-primary">{leaf?.categoryName}</span> خود را وارد کنید <span className="text-red-500">*</span>
   359→            </Label>
   360→            <Input
   361→              id="product-type"
   362→              value={draft.productType}
   363→              onChange={(e) => patch({ productType: e.target.value })}
   364→              placeholder={leaf?.example ? `مثلا : ${leaf.example}` : 'مثلا : اکبری'}
   365→            />
   366→            <p className="text-[11px] text-muted-foreground">
   367→              {leaf?.example ? `مثلا : ${leaf.example}` : 'نوع/واریته دقیق کالا را بنویسید تا خریداران مرتبط پیدا شوید.'}
   368→            </p>
   369→          </div>
   370→
   371→          <div className="grid gap-4 sm:grid-cols-2">
   372→            <div className="space-y-2">
   373→              <Label>
   374→                استان مبدا <span className="text-red-500">*</span>
   375→              </Label>
   376→              <Select value={draft.provinceId} onValueChange={(v) => patch({ provinceId: v, cityId: '' })}>
   377→                <SelectTrigger className="w-full">
   378→                  <SelectValue placeholder="انتخاب استان" />
   379→                </SelectTrigger>
   380→                <SelectContent>
   381→                  {(provinces ?? []).map((p) => (
   382→                    <SelectItem key={p.id} value={String(p.id)}>
   383→                      {p.name}
   384→                    </SelectItem>
   385→                  ))}
   386→                </SelectContent>
   387→              </Select>
   388→            </div>
   389→            <div className="space-y-2">
   390→              <Label>
   391→                شهر مبدا <span className="text-red-500">*</span>
   392→              </Label>
   393→              <Select value={draft.cityId} onValueChange={(v) => patch({ cityId: v })} disabled={!draft.provinceId}>
   394→                <SelectTrigger className="w-full">
   395→                  <SelectValue placeholder={draft.provinceId ? 'انتخاب شهر' : 'ابتدا استان را انتخاب کنید'} />
   396→                </SelectTrigger>
   397→                <SelectContent>
   398→                  {(cities ?? []).map((c) => (
   399→                    <SelectItem key={c.id} value={String(c.id)}>
   400→                      {c.name}
   401→                    </SelectItem>
   402→                  ))}
   403→                </SelectContent>
   404→              </Select>
   405→            </div>
   406→          </div>
   407→        </PanelCard>
   408→      )}
   409→
   410→      {/* ─── مرحله ۳: موجودی و قیمت ─── */}
   411→      {step === 2 && (
   412→        <PanelCard className="space-y-6">
   413→          <div className="space-y-2.5">
   414→            <Label htmlFor="stock">
   415→              میزان موجودی محصول <span className="text-muted-foreground">({leaf?.unit ?? 'کیلوگرم'})</span>{' '}
   416→              <span className="text-red-500">*</span>
   417→            </Label>
   418→            <UnitInput
   419→              id="stock"
   420→              kind="weight"
   421→              value={draft.stock}
   422→              onValueChange={(v) => patch({ stock: v })}
   423→              unit={leaf?.unit ?? 'کیلوگرم'}
   424→            />
   425→            <p className="text-[11px] text-muted-foreground">مقیاس عمده — مقیاس‌های خرده‌فروشی قابل قبول نیستند.</p>
   426→          </div>
   427→          <div className="space-y-2.5">
   428→            <Label htmlFor="min-amount">
   429→              حداقل سفارش <span className="text-muted-foreground">({leaf?.unit ?? 'کیلوگرم'})</span>{' '}
   430→              <span className="text-red-500">*</span>
   431→            </Label>
   432→            <UnitInput
   433→              id="min-amount"
   434→              kind="weight"
   435→              value={draft.minSaleAmount}
   436→              onValueChange={(v) => patch({ minSaleAmount: v })}
   437→              unit={leaf?.unit ?? 'کیلوگرم'}
   438→            />
   439→          </div>
