   106→   106→/** خواندن ?thread= — «support» یا ObjectId؛ چیز دیگر نادیده گرفته می‌شود */
   107→   107→function threadFromUrl(): string | null {
   108→   108→  try {
   109→   109→    const t = new URLSearchParams(window.location.search).get('thread');
   110→   110→    if (t === SUPPORT_URL_VALUE) return SUPPORT_ID;
   111→   111→    return t && /^[0-9a-fA-F]{24}$/.test(t) ? t : null;
   112→   112→  } catch {
   113→   113→    return null;
   114→   114→  }
   115→   115→}

   118→   118→ *  رفرش همان چت را باز می‌کند و back مرورگر به لیست برمی‌گردد. pushState
   119→   119→ *  (نه replace) تا هر مکالمه در تاریخچهٔ مرورگر جای خودش را داشته باشد. */
   120→   120→function syncThreadUrl(id: string | null) {
   121→   121→  try {
   122→   122→    const url = new URL(window.location.href);
   123→   123→    if (id) url.searchParams.set('thread', id === SUPPORT_ID ? SUPPORT_URL_VALUE : id);
   124→   124→    else url.searchParams.delete('thread');
   125→   125→    window.history.pushState(null, '', url);
   126→   126→  } catch {
   127→   127→    /* محیط بدون history — بی‌خطر */
   128→   128→  }
   129→   129→}
