     4→// اصول:
     5→//  • نام فیلدها در دیتابیس snake_case عین باسکول (product_name, min_sale_price,
     6→//    province_id …) تا داده‌های کراول‌شده از باسکول بدون تبدیل اینزرت شوند.
     7→//  • داده‌های پایه (Category/Province/City/Product) id عددی باسکول دارند
     8→//    (پسته=44، اصفهان=4، شهر بابک=230) — Prisma روی مونگو id صریح عددی را
     9→//    پشتیبانی می‌کند؛ شمارنده جدول Counter برای id محصولات جدید.
    10→//  • تصاویر هر مدل از طریق جدول چندریختی File مدیریت می‌شوند (دکترین
    11→//    imach-back): fieldKey اسلات را تعیین می‌کند، ردیف staged قبل از چسبیدن
    12→//    به مدل توسط reaper حذف می‌شود و replace اول آپلود می‌کند بعد قبلی را
    13→//    پاک می‌کند — هیچ فایل سرگردانی روی حافظه گران نمی‌ماند.
    14→//  • TTL ایندکس در Prisma پشتیبانی نمی‌شود → پاک‌سازی OTP/RefreshToken با
    15→//    reaper دوره‌ای درون پروسه انجام می‌شود.

   260→/// ─── لوکیشن درختی واحد ───────────────────────────────────────────────────────
   261→/// به‌جای دو جدول جداگانهٔ استان/شهر، یک کالکشن واحد با ساختار درختی:
   262→///   province ← city ← region ← neighborhood …
   263→/// «فقط یک جدول لوکیشن که فقط تایپش متفاوت باشه» — خواستهٔ مالک؛ سطح‌های
   264→/// آینده (منطقه/محله) بدون مهاجرت، فقط با type جدید + parent_id اضافه می‌شوند.
   265→///
   266→/// ⚠️ چرا id رشتهٔ ترکیبی است؟ چون id های عددی باسکول در دو فضای جدا هستند و
   267→/// برخورد عددی دارند (شهر ۴ = بستان‌آبادِ آذربایجان شرقی، استان ۴ = اصفهان) —
   268→/// یعنی «عدد + نوع» با هم یکتا هستند. کلید یکتا: «province:4» / «city:230».
   269→/// عدد خام باسکول در buskolId حفظ می‌شود و در API ها همان id عددی برمی‌گردد؛
   270→/// آگهی‌ها/درخواست‌ها به همان اعداد خام اشاره می‌کنند (بدون تغییر).
   271→model Location {
   272→  /// کلید یکتای سراسری — «province:4» | «city:230» | «region:…"
   273→  id        String  @id @map("_id")
   274→  /// id عددی باسکول در نوع خودش — یکتا فقط با هم type
   275→  buskolId  Int     @map("buskool_id")
   276→  /// province | city | region | neighborhood
   277→  type      String
   278→  /// کلید گره والد («province:4») — null یعنی ریشه (استان)
   279→  parentId  String? @map("parent_id")
   280→  /// نام فارسی
   281→  name      String
   282→  enName    String? @map("en_name")
   283→  latitude  Float?
   284→  longitude Float?
   285→  /// فعال/غیرفعال (A2) — مکان غیرفعال از لیست‌های عمومی پنهان می‌شود و
   286→  /// ثبت آگهی/درخواست جدید روی آن ممکن نیست؛ آگهی‌های قدیمی نامشان را همچنان resolve می‌کنند
   287→  isActive  Boolean @default(true) @map("is_active")
   288→
   289→  @@unique([type, buskolId])
   290→  @@index([type, parentId])
   291→  @@index([type, name])
   292→}

   296→/// جدول چندریختی File — تصویر/پیوست هر مدلی با fieldKey مشخص می‌شود:
   297→///   User.avatar · Product.gallery · (آینده: ChargeOrder.receipt, Verification.docs)
   298→/// ردیف staged (relatedId=null) اگر تا TTL به مدلی نچسبد → reaper حذفش می‌کند.
   299→/// ⚠️ relatedId رشته ساده است (نه ObjectId) — چندریختی: id کاربر ObjectId است
   300→/// اما id محصول عدد باسکول؛ هر دو به شکل رشته ذخیره/کوئری می‌شوند.
   301→model File {
   302→  id          String   @id @default(auto()) @map("_id") @db.ObjectId
   303→  ownerId     String   @map("owner_id") @db.ObjectId
   304→  owner       User     @relation("FileOwner", fields: [ownerId], references: [id], onDelete: Cascade)
   305→  /// نام مدل مقصد — فقط مقادیر RELATED_MODELS سرویس
   306→  relatedModel String  @map("related_model")
   307→  /// id مدل مقصد (رشته) — null یعنی staged
   308→  relatedId   String?  @map("related_id")
   309→  /// اسلات: avatar | gallery | receipt | …
   310→  fieldKey    String   @map("field_key")
   311→  /// توضیح اختیاری
   312→  description String?
   313→  /// نام فایل اصلی (فقط نمایش)
   314→  name        String
   315→  mimeType    String   @map("mime_type")
   316→  /// حجم فایل اصلی بعد از فشرده‌سازی (بایت)
   317→  size        Int
   318→  /// URL عمومی مستقیم — بدون توکن، از CDN
   319→  url         String
   320→  thumbUrl    String?  @map("thumb_url")
   321→  /// کلید ذخیره‌سازی فایل اصلی (دستگیره حذف/تعویض پروایدر) — null برای URL خارجی
   322→  storageKey  String?  @map("storage_key")
   323→  thumbStorageKey String? @map("thumb_storage_key")
   324→  /// متادیتا (ابعاد، حجم اصلی، …)
