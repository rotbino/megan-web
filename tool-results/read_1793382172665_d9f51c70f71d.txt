   267→/// بود — نوع کالا ورودی آزاد کاربران است، نه دسته. حالا جدول جدا دارد:
   268→///  • در هیچ ناوبری دسته‌ای (سایدبار/مگامنو/کشکید) دیده نمی‌شود؛
   269→///  • هنگام ثبت/ویرایش درخواست خرید idempotent ساخته/بازمصرف می‌شود؛
   270→///  • برای تطبیق «خریداران پیشنهادی» و آمار استفاده می‌شود.
   271→model Good {
   272→  /// id عددی از Counter('good') — مهاجرت: همان id نودهای PRODUCT_TYPE قدیمی

   280→  /// دفعات استفاده در درخواست‌های خرید
   281→  usageCount  Int      @default(0) @map("usage_count")
   282→
   283→  createdAt DateTime @default(now()) @map("created_at")
   284→  updatedAt DateTime @updatedAt @map("updated_at")
   285→
   286→  @@unique([categoryId, name])
   287→  @@index([categoryId])
   288→  @@index([name])
   289→}

   291→/// ─── لوکیشن درختی واحد ───────────────────────────────────────────────────────
   292→/// به‌جای دو جدول جداگانهٔ استان/شهر، یک کالکشن واحد با ساختار درختی:
   293→///   province ← city ← region ← neighborhood …
   294→/// «فقط یک جدول لوکیشن که فقط تایپش متفاوت باشه» — خواستهٔ مالک؛ سطح‌های
   295→/// آینده (منطقه/محله) بدون مهاجرت، فقط با type جدید + parent_id اضافه می‌شوند.
   296→///
   297→/// ⚠️ چرا id رشتهٔ ترکیبی است؟ چون id های عددی باسکول در دو فضای جدا هستند و
   298→/// برخورد عددی دارند (شهر ۴ = بستان‌آبادِ آذربایجان شرقی، استان ۴ = اصفهان) —
   299→/// یعنی «عدد + نوع» با هم یکتا هستند. کلید یکتا: «province:4» / «city:230».
   300→/// عدد خام باسکول در buskolId حفظ می‌شود و در API ها همان id عددی برمی‌گردد؛
   301→/// آگهی‌ها/درخواست‌ها به همان اعداد خام اشاره می‌کنند (بدون تغییر).
   302→model Location {
   303→  /// کلید یکتای سراسری — «province:4» | «city:230» | «region:…"
   304→  id        String  @id @map("_id")
   305→  /// id عددی باسکول در نوع خودش — یکتا فقط با هم type
   306→  buskolId  Int     @map("buskool_id")
   307→  /// province | city | region | neighborhood
   308→  type      String
   309→  /// کلید گره والد («province:4») — null یعنی ریشه (استان)
   310→  parentId  String? @map("parent_id")
   311→  /// نام فارسی
   312→  name      String
   313→  enName    String? @map("en_name")
   314→  latitude  Float?
   315→  longitude Float?
   316→  /// فعال/غیرفعال (A2) — مکان غیرفعال از لیست‌های عمومی پنهان می‌شود و
   317→  /// ثبت آگهی/درخواست جدید روی آن ممکن نیست؛ آگهی‌های قدیمی نامشان را همچنان resolve می‌کنند
