    64→/** دیالوگ اصلاح کیف پول — شارژ/کسر با دلیل */
    65→function WalletDialog({
    66→  userId,
    67→  balance,
    68→  onClose,
    69→}: {
    70→  userId: string;
    71→  balance: number;
    72→  onClose: () => void;
    73→}) {
    74→  const { toast } = useToast();
    75→  const queryClient = useQueryClient();
    76→  const [mode, setMode] = useState<'charge' | 'deduct'>('charge');
    77→  const [amount, setAmount] = useState('');
    78→  const [reason, setReason] = useState('');
    79→  const [error, setError] = useState('');
    80→
    81→  const save = useMutation({
    82→    mutationFn: () => {
    83→      const n = Math.abs(parseInt(amount.replace(/[^\d]/g, ''), 10) || 0);
    84→      return creditAdminUserWallet(userId, mode === 'charge' ? n : -n, reason.trim());
    85→    },
    86→    onSuccess: (d) => {
    87→      void queryClient.invalidateQueries({ queryKey: ['admin-user', userId] });
    88→      toast({ description: `کیف پول اصلاح شد — موجودی جدید: ${priceFa(d.balance)}` });
    89→      onClose();
    90→    },
    91→    onError: (err: Error) => setError(err.message),
    92→  });
    93→
    94→  const submit = (e: React.FormEvent) => {
    95→    e.preventDefault();
    96→    setError('');
    97→    const n = Math.abs(parseInt(amount.replace(/[^\d]/g, ''), 10) || 0);
    98→    if (n <= 0) {
    99→      setError('مبلغ معتبر وارد کنید.');
   100→      return;
   101→    }
   102→    if (reason.trim().length < 3) {
   103→      setError('دلیل اصلاح را بنویسید (برای کاربر نمایش داده می‌شود).');
   104→      return;
   105→    }
   106→    save.mutate();
   107→  };
   108→
   109→  return (
   110→    <div className="fixed inset-0 z-50 grid place-items-center p-4" role="dialog" aria-modal="true" aria-label="اصلاح کیف پول">
   111→      <button type="button" aria-label="بستن" className="absolute inset-0 bg-black/40" onClick={onClose} />
   112→      <form onSubmit={submit} className="relative w-full max-w-sm space-y-4 rounded-xl border bg-card p-6 shadow-lg">
   113→        <div>
   114→          <h2 className="text-sm font-extrabold">اصلاح کیف پول</h2>
   115→          <p className="mt-1 text-[11px] text-muted-foreground">
   116→            موجودی فعلی: {priceFa(balance)} — اصلاح با دلیل در تاریخچه کاربر ثبت می‌شود.
   117→          </p>
   118→        </div>
   119→
   120→        {/* شارژ / کسر */}
   121→        <div className="grid grid-cols-2 gap-2">
   122→          {(['charge', 'deduct'] as const).map((m) => (
   123→            <button
   124→              key={m}
   125→              type="button"
   126→              onClick={() => setMode(m)}
   127→              className={cn(
   128→                'flex h-10 items-center justify-center gap-1.5 rounded-[10px] border text-[12.5px] font-bold transition-colors',
   129→                mode === m
   130→                  ? 'border-primary-line bg-primary-soft text-primary'
   131→                  : 'border-input bg-card text-secondary-foreground hover:bg-secondary/60',
   132→              )}
   133→            >
   134→              {m === 'charge' ? <Plus size={14} /> : <Minus size={14} />}
   135→              {m === 'charge' ? 'شارژ دستی' : 'کسر دستی'}
   136→            </button>
   137→          ))}
   138→        </div>
   139→
   140→        <div className="space-y-1.5">
   141→          <Label htmlFor="wa-amount">مبلغ (تومان)</Label>
   142→          <Input
   143→            id="wa-amount"
   144→            inputMode="numeric"
   145→            dir="ltr"
   146→            value={amount}
   147→            onChange={(e) => setAmount(e.target.value)}
   148→            placeholder="مثلاً 500000"
   149→            maxLength={12}
   150→            autoFocus
   151→          />
   152→        </div>
   153→
   154→        <div className="space-y-1.5">
   155→          <Label htmlFor="wa-reason">دلیل (برای کاربر نمایش داده می‌شود)</Label>
   156→          <Textarea
   157→            id="wa-reason"
   158→            value={reason}
   159→            onChange={(e) => setReason(e.target.value)}
   160→            placeholder="مثلاً: جبران واریز حضوری شعبه"
   161→            maxLength={200}
   162→            className="min-h-20"
   163→          />
   164→        </div>
   165→
   166→        {error && <p className="text-xs text-destructive">{error}</p>}
   167→
   168→        <div className="flex gap-2 pt-1">
   169→          <Button type="submit" className="flex-1 rounded-md" disabled={save.isPending}>
   170→            {save.isPending ? <Loader2 className="animate-spin" size={16} /> : null}
   171→            اعمال اصلاح
   172→          </Button>
   173→          <Button type="button" variant="outline" className="rounded-md" onClick={onClose}>
   174→            انصراف
   175→          </Button>
   176→        </div>
   177→      </form>
   178→    </div>
   179→  );
   180→}

   182→/** دیالوگ رد مدارک احراز با دلیل */
   183→function RejectDialog({ orderId, onClose }: { orderId: string; onClose: () => void }) {
   184→  const { toast } = useToast();
   185→  const queryClient = useQueryClient();
   186→  const [reason, setReason] = useState('');
   187→  const [error, setError] = useState('');
   188→
   189→  const reject = useMutation({
   190→    mutationFn: () => reviewAdminVerification(orderId, false, reason.trim()),
   191→    onSuccess: () => {
   192→      void queryClient.invalidateQueries({ queryKey: ['admin-user'] });
   193→      toast({ description: 'درخواست احراز رد شد — دلیل برای کاربر ثبت شد.' });
   194→      onClose();
   195→    },
   196→    onError: (err: Error) => setError(err.message),
   197→  });
   198→
   199→  const submit = (e: React.FormEvent) => {
   200→    e.preventDefault();
   201→    setError('');
   202→    if (reason.trim().length < 3) {
   203→      setError('دلیل رد را بنویسید.');
   204→      return;
   205→    }
   206→    reject.mutate();
   207→  };
   208→
   209→  return (
   210→    <div className="fixed inset-0 z-50 grid place-items-center p-4" role="dialog" aria-modal="true" aria-label="رد مدارک احراز">
   211→      <button type="button" aria-label="بستن" className="absolute inset-0 bg-black/40" onClick={onClose} />
   212→      <form onSubmit={submit} className="relative w-full max-w-sm space-y-4 rounded-xl border bg-card p-6 shadow-lg">
   213→        <div>
   214→          <h2 className="text-sm font-extrabold">رد مدارک احراز هویت</h2>
   215→          <p className="mt-1 text-[11px] text-muted-foreground">دلیل برای کاربر نمایش داده می‌شود تا مدارک درست را دوباره بفرستد.</p>
   216→        </div>
   217→        <div className="space-y-1.5">
   218→          <Label htmlFor="rj-reason">دلیل رد</Label>
   219→          <Textarea
   220→            id="rj-reason"
   221→            value={reason}
   222→            onChange={(e) => setReason(e.target.value)}
   223→            placeholder="مثلاً: تصویر کارت ملی ناخواناست"
   224→            maxLength={300}
   225→            className="min-h-20"
   226→            autoFocus
   227→          />
   228→        </div>
   229→        {error && <p className="text-xs text-destructive">{error}</p>}
   230→        <div className="flex gap-2 pt-1">
   231→          <Button type="submit" variant="destructive" className="flex-1 rounded-md" disabled={reject.isPending}>
   232→            {reject.isPending ? <Loader2 className="animate-spin" size={16} /> : null}
   233→            رد مدارک
   234→          </Button>
   235→          <Button type="button" variant="outline" className="rounded-md" onClick={onClose}>
   236→            انصراف
   237→          </Button>
   238→        </div>
   239→      </form>
   240→    </div>
   241→  );
   242→}

   613→                  </div>
   614→
   615→                  {data.verification.status === 'pending' && (

   655→                              اندازه کامل ↗

   658→                        </div>
   659→                        {doc ? (
   660→                           
   661→                          <img
   662→                            src={doc.thumbUrl ?? doc.url}
   663→                            alt={label}
   664→                            loading="lazy"
   665→                            className="h-44 w-full bg-secondary object-cover"

   685→      </Tabs>
   686→
   687→      {walletOpen ? <WalletDialog userId={id} balance={u.wallet_balance} onClose={() => setWalletOpen(false)} /> : null}
   688→      {rejectOpen && verificationOrderId ? (
