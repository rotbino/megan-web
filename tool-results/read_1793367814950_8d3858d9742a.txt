    76→const toProduct = (p: RawProduct): Product => {
    77→  const { status, ...rest } = p;
    78→  return {
    79→    ...rest,
    80→    minSalePrice: p.minSalePrice ?? null,
    81→    photoThumbs: p.photoThumbs ?? p.photos,
    82→    seller: {
    83→      ...p.seller,

    87→    // نمی‌فرستند — پیش‌فرض confirmed تا تب‌های «محصولات من» درست فیلتر شوند
    88→    status: (status ?? 'confirmed') as Product['status'],
    89→  };

   416→/** تایید کد + ورود — توکن‌ها را ذخیره و کاربر نشست را برمی‌گرداند
   417→ *  (ساختار برای پنل پیامک آینده — فعلاً فرانت از loginWithSession استفاده می‌کند) */
   418→export async function verifyOtp(input: {
   419→  phone: string;
   420→  code: string;
   421→}): Promise<{ ok: boolean; isNewUser?: boolean; user?: SessionUser }> {
   422→  const data = await apiFetch<{
   423→    isNewUser: boolean;
   424→    accessToken: string;
   425→    refreshToken: string;
   426→  }>('/auth/verifyOtp', input, { auth: false, retryOn401: false });
   427→
   428→  setTokens({ accessToken: data.accessToken, refreshToken: data.refreshToken });
   429→
   430→  // کاربر نشست — از پروفایل واقعی؛ اگر موقتاً نرسید، حداقلِ قابل‌نمایش
   431→  let user: SessionUser = {
   432→    phone: input.phone,
   433→    userName: '',
   434→    firstName: '',
   435→    lastName: '',
   436→    avatarUrl: null,
   437→    role: 'buyer',
   438→    hasPassword: false,
   439→  };
   440→  try {
   441→    const profile = await apiFetch<{ user: RawProfileUser }>('/users/getProfile', {});
   442→    user = mapSessionUser(profile.user);
   443→  } catch {
   444→    /* پروفایل اختیاری است — ورود موفق بوده */
   445→  }
   446→
   447→  return { ok: true, isNewUser: data.isNewUser, user };
   448→}

   450→/**
   451→ * ورود روش سشن — فقط با شماره موبایل (بدون پیامک، تا آمدن پنل پیامک).
   452→ * • کاربر بدون رمز → مستقیم وارد می‌شود (مثل دفعه اول).
   453→ * • کاربر رمزدار → بک‌اند PASSWORD_REQUIRED می‌دهد؛ requiresPassword=true
   454→ *   برمی‌گردد تا UI فیلد رمز را نشان دهد.
   455→ * کاربر جدید → isNewUser برای هدایت به /register?step=info.
   456→ */
   457→export async function loginWithSession(input: {
   458→  phone: string;
   459→  password?: string;
   460→}): Promise<{
   461→  ok: boolean;
   462→  requiresPassword?: boolean;
   463→  isNewUser?: boolean;
   464→  user?: SessionUser;
   465→}> {
   466→  let data: { isNewUser: boolean; hasPassword: boolean; accessToken: string; refreshToken: string };
   467→  try {
   468→    data = await apiFetch('/auth/login', input, { auth: false, retryOn401: false });
   469→  } catch (err) {
   470→    if (err instanceof ApiError && err.code === 'PASSWORD_REQUIRED') {
   471→      return { ok: false, requiresPassword: true };
   472→    }
   473→    throw err;
   474→  }
   475→
   476→  setTokens({ accessToken: data.accessToken, refreshToken: data.refreshToken });
   477→
   478→  // کاربر نشست — از پروفایل واقعی؛ اگر موقتاً نرسید، حداقلِ قابل‌نمایش
   479→  let user: SessionUser = {
   480→    phone: input.phone,
   481→    userName: '',
   482→    firstName: '',
   483→    lastName: '',
   484→    avatarUrl: null,
   485→    role: 'buyer',
   486→    hasPassword: data.hasPassword,
   487→  };
   488→  try {
   489→    const profile = await apiFetch<{ user: RawProfileUser }>('/users/getProfile', {});
   490→    user = mapSessionUser(profile.user);
   491→  } catch {
   492→    /* پروفایل اختیاری است — ورود موفق بوده */
   493→  }
   494→
   495→  return { ok: true, isNewUser: data.isNewUser, user };
   496→}

   506→/** نگاشت user بک‌اند → کاربر نشست فرانت */
   507→export function mapSessionUser(u: RawProfileUser): SessionUser {
   508→  return {
   509→    phone: u.phone,
   510→    userName: u.user_name ?? '',
   511→    firstName: u.first_name ?? '',
   512→    lastName: u.last_name ?? '',
   513→    avatarUrl: u.avatar_url ?? null,
   514→    // پیش‌فرض فروشنده (خواسته مالک) — فقط وقتی خودش خریدار را انتخاب کرده باشد
   515→    // (is_buyer=true و is_seller=false) پنل خریدار باز می‌شود؛ انتخاب در بک‌اند ماندگار است.
   516→    role: u.is_buyer && !u.is_seller ? 'buyer' : 'seller',
   517→    hasPassword: Boolean(u.has_password),
   518→  };
   519→}

   521→/** کاربر نشست از پروفایل واقعی — null یعنی لاگین نیست؛ خطای شبکه reject می‌شود */
   522→export async function fetchAuthUser(): Promise<SessionUser | null> {
   523→  if (!getTokens()?.accessToken) return null;
   524→  try {
   525→    const profile = await apiFetch<{ user: RawProfileUser }>('/users/getProfile', {});
   526→    return mapSessionUser(profile.user);
   527→  } catch (err) {
   528→    if (err instanceof ApiError && (err.status === 401 || err.status === 403)) return null;

   684→  stock: number;
   685→  minSaleAmount: number;

   801→    } catch {
   802→      failedPhotos.push(file);
   803→    }
   804→  }
   805→
   806→  return { ok: true, id: updated.id, status: 'pending', failedPhotos };

  1019→
  1020→  const res = await fetch('/api/verification/submitVerification', {
  1021→    method: 'POST',
  1022→    headers: { Authorization: `Bearer ${tokens.accessToken}` },
  1023→    body: form,
  1024→  });
  1025→  const json = (await res.json().catch(() => null)) as
  1026→    | { status: true; data: MyVerification; message?: string }
  1027→    | { status: false; message: string; code?: string }
  1028→    | null;
  1029→  if (!json) throw new ApiError('ارسال مدارک ناموفق بود.', res.status, 'NETWORK');
  1030→  if (json.status === false) throw new ApiError(json.message, res.status, json.code);

  1106→/** ارسال فیش شارژ کارت‌به‌کارت — واقعی: multipart (packageKey + receipt) */
  1107→export async function submitChargeRequest(input: {
  1108→  packageKey: ChargePackage['key'];
  1109→  receiptDataUrl: string;
  1110→}): Promise<{ ok: true; transactionId: string }> {
  1111→  const tokens = getTokens();
  1112→  if (!tokens?.accessToken) {
  1113→    throw new ApiError('برای شارژ کیف پول ابتدا وارد حساب خود شوید.', 401, 'AUTH_REQUIRED');
  1114→  }
  1115→
  1116→  // dataURL → Blob (فرم صفحه همان dataURL را می‌دهد)
  1117→  const form = new FormData();
  1118→  form.append('packageKey', input.packageKey);
  1119→  form.append('receipt', dataUrlToBlob(input.receiptDataUrl));
  1120→
  1121→  const res = await fetch('/api/wallet/createChargeOrder', {
  1122→    method: 'POST',
  1123→    headers: { Authorization: `Bearer ${tokens.accessToken}` },
  1124→    body: form,
  1125→  });
  1126→  const json = (await res.json().catch(() => null)) as
  1127→    | { status: true; data?: { id?: number }; message?: string }
  1128→    | { status: false; message: string; code?: string }
  1129→    | null;
  1130→  if (!json) throw new ApiError('ارسال فیش ناموفق بود.', res.status, 'NETWORK');
  1131→  if (json.status === false) throw new ApiError(json.message, res.status, json.code);
  1132→  return { ok: true, transactionId: String(json.data?.id ?? '') };
  1133→}

  1135→/** dataURL → Blob برای فرم multipart */
  1136→function dataUrlToBlob(dataUrl: string): Blob {
  1137→  const [head, body] = dataUrl.split(',');
  1138→  const mime = /data:(.*?)(;|$)/.exec(head)?.[1] || 'image/jpeg';
  1139→  const bin = atob(body);
  1140→  const arr = new Uint8Array(bin.length);
  1141→  for (let i = 0; i < bin.length; i++) arr[i] = bin.charCodeAt(i);
  1142→  return new Blob([arr], { type: mime });
  1143→}
