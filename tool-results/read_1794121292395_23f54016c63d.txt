   101→/** مقدار پشتیبانی در URL — از id های ۲۴کاراکتری MongoDB جدا می‌ماند */
   102→const SUPPORT_URL_VALUE = 'support';
   103→
   104→/** خواندن ?thread= — «support» یا ObjectId؛ چیز دیگر نادیده گرفته می‌شود */
   105→function threadFromUrl(): string | null {
   106→  try {
   107→    const t = new URLSearchParams(window.location.search).get('thread');
   108→    if (t === SUPPORT_URL_VALUE) return SUPPORT_ID;
   109→    return t && /^[0-9a-fA-F]{24}$/.test(t) ? t : null;
   110→  } catch {
   111→    return null;
   112→  }
   113→}

   115→/** مکالمهٔ جاری را در URL نگه می‌دارد (خواسته مالک ۸-۱۱):
   116→ *  رفرش همان چت را باز می‌کند و back مرورگر به لیست برمی‌گردد. pushState
   117→ *  (نه replace) تا هر مکالمه در تاریخچهٔ مرورگر جای خودش را داشته باشد. */
   118→function syncThreadUrl(id: string | null) {
   119→  try {
   120→    const url = new URL(window.location.href);
   121→    if (id) url.searchParams.set('thread', id === SUPPORT_ID ? SUPPORT_URL_VALUE : id);
   122→    else url.searchParams.delete('thread');
   123→    window.history.pushState(null, '', url);
   124→  } catch {
   125→    /* محیط بدون history — بی‌خطر */
   126→  }
   127→}
