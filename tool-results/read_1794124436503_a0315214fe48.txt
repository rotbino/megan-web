   103→/** مقدار پشتیبانی در URL — از id های ۲۴کاراکتری MongoDB جدا می‌ماند */

   106→/** خواندن ?thread= — «support» یا ObjectId؛ چیز دیگر نادیده گرفته می‌شود */
   107→function threadFromUrl(): string | null {
   108→  try {
   109→    const t = new URLSearchParams(window.location.search).get('thread');
   110→    if (t === SUPPORT_URL_VALUE) return SUPPORT_ID;
   111→    return t && /^[0-9a-fA-F]{24}$/.test(t) ? t : null;
   112→  } catch {
   113→    return null;
   114→  }
   115→}

   117→/** مکالمهٔ جاری را در URL نگه می‌دارد (خواسته مالک ۸-۱۱):
   118→ *  رفرش همان چت را باز می‌کند و back مرورگر به لیست برمی‌گردد. pushState
   119→ *  (نه replace) تا هر مکالمه در تاریخچهٔ مرورگر جای خودش را داشته باشد. */
   120→function syncThreadUrl(id: string | null) {
   121→  try {
   122→    const url = new URL(window.location.href);
   123→    if (id) url.searchParams.set('thread', id === SUPPORT_ID ? SUPPORT_URL_VALUE : id);
   124→    else url.searchParams.delete('thread');
   125→    window.history.pushState(null, '', url);
   126→  } catch {
   127→    /* محیط بدون history — بی‌خطر */
   128→  }
   129→}

   541→                    <span className="mt-0.5 flex items-center justify-between gap-2">

   543→                        {supportPreview}

   635→                    صفحهٔ پشتیبانی

   651→                  <ArrowRight size={17} />

   726→/** ردیف مکالمه در لیست — بج خوانده‌نشده قرمز (خواسته مالک ۷-۱۱) */

   811→/**
   812→ * پیام لینک‌دارِ ماندگار پاسخ پشتیبان — خواستهٔ مالک ۸-۱۱ دور ۲۴:
   813→ * «هیچ پیامی داخل پشتیبانی ایمچ نیست که شامل لینک پنل پشتیبانی باشد».
   814→ * هر دستهٔ پاسخ پشتیبان در گفتگوی تیکت = یک پیام «شما N پیام در پنل
   815→ * پشتیبانی دارید» + لینک «مشاهدهٔ گفتگو با پشتیبان» → /panel/support.
   816→ * حلقهٔ کامل: بج هدر → لیست چت (بج روی پشتیبانی ایمچ) → همین پیام → تیکت.
   817→ * ماندگار است — خواندن تیکت فقط رنگ قرمز را آرام می‌کند؛ پیام در
   818→ * تاریخچهٔ چت می‌ماند (برخلاف کارت/نوار موقتِ شمارنده‌محور).
   819→ */
   820→function SupportReplyNotice({ messages }: { messages: SupportChatMessage[] }) {
   821→  const last = messages[messages.length - 1];
   822→  const unread = messages.some((m) => m.readAt === null);
   823→  const count = messages.length;
   824→  // پیش‌نمایش یک‌خطی آخرین پاسخ — کارت‌های تاریخی از هم قابل تشخیص شوند
   825→  const previewOf = (m: SupportChatMessage) =>
   826→    m.text.trim() ||
   827→    (m.kind === 'image' ? '📷 تصویر' : m.kind === 'video' ? '🎬 ویدئو' : m.kind === 'file' ? '📎 فایل' : '');
   828→  return (
   829→    <div className="flex w-full justify-end">
   830→      <Link
   831→        href="/panel/support"
   832→        className={cn(
   833→          'group block w-fit max-w-[85%] rounded-2xl rounded-se-md border px-3.5 py-2.5 shadow-sm transition-colors sm:max-w-[75%] sm:px-4',
   834→          'animate-in fade-in slide-in-from-bottom-1 duration-200',
   835→          unread
   836→            ? 'border-red-200 bg-red-50 hover:bg-red-100'
   837→            : 'border-border/70 bg-card hover:bg-secondary/60',
   838→        )}
   839→      >
   840→        <p
   841→          className={cn(
   842→            'text-[13px] font-bold leading-6',
   843→            unread ? 'text-red-700' : 'text-foreground',
   844→          )}
   845→        >
   846→          شما {count === 1 ? 'یک' : toFa(count)} پیام در پنل پشتیبانی دارید
   847→        </p>
   848→        {previewOf(last) && (
   849→          <p dir="auto" className="mt-0.5 line-clamp-1 max-w-60 text-[11px] leading-5 text-muted-foreground">
   850→            آخرین پاسخ: {previewOf(last)}
   851→          </p>
   852→        )}
   853→        <p className="mt-0.5 text-[11px] font-bold leading-5 text-primary underline underline-offset-2">
   854→          مشاهدهٔ گفتگو با پشتیبان
   855→        </p>
   856→        <p className="mt-1.5 flex items-center gap-1.5 text-[9px] text-muted-foreground">
   857→          <span>پشتیبانی ایمچ</span>
   858→          {last.adminUsername ? (
   859→            <>
   860→              ·<span>{last.adminUsername}</span>
   861→            </>
   862→          ) : null}
   863→          ·<span><RelativeTime date={last.createdAt} /></span>
   864→        </p>
   865→      </Link>
   866→    </div>
   867→  );
   868→}
