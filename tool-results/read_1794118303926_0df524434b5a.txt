   110→function isServer(): boolean {
   111→  return typeof window === 'undefined';
   112→}
   113→
   114→export function getAdminToken(): string | null {
   115→  if (isServer()) return null;
   116→  try {
   117→    return localStorage.getItem(ADMIN_TOKEN_KEY);
   118→  } catch {
   119→    return null;
   120→  }
   121→}
   122→
   123→export function setAdminToken(token: string): void {
   124→  if (isServer()) return;
   125→  try {
   126→    localStorage.setItem(ADMIN_TOKEN_KEY, token);
   127→  } catch {
   128→    /* حافظه پر است — فقط نشست جاری */
   129→  }
   130→}
   131→
   132→export function clearAdminToken(): void {
   133→  if (isServer()) return;
   134→  try {
   135→    localStorage.removeItem(ADMIN_TOKEN_KEY);
   136→  } catch {
   137→    /* noop */
   138→  }
   139→}

   141→// ─── فراخوانی ────────────────────────────────────────────────────────────────
   142→
   143→function apiBase(): string {
   144→  return isServer()
   145→    ? `${process.env.BACKEND_URL ?? 'http://localhost:4000'}/api/v1`
   146→    : '/api';
   147→}
   148→
   149→export async function adminFetch<T>(path: string, body?: unknown, opts: { auth?: boolean } = {}): Promise<T> {
   150→  const needAuth = opts.auth !== false;
   151→  const token = getAdminToken();
   152→
   153→  const res = await fetch(`${apiBase()}${path.startsWith('/') ? path : `/${path}`}`, {
   154→    method: 'POST',
   155→    headers: {
   156→      'Content-Type': 'application/json',
   157→      ...(needAuth && token ? { Authorization: `Bearer ${token}` } : {}),
   158→    },
   159→    body: JSON.stringify(body ?? {}),
   160→    cache: 'no-store',
   161→  });
   162→
   163→  const json = (await res.json().catch(() => null)) as
   164→    | { status: true; data: T; message?: string }
   165→    | { status: false; message: string; code?: string }
   166→    | null;
   167→
   168→  if (!json) throw new ApiError('ارتباط با سرور برقرار نشد.', res.status, 'NETWORK');
   169→  if (json.status === false) {
   170→    // نشست ادمین تمام شده — توکن خراب فوراً پاک شود
   171→    if (res.status === 401) clearAdminToken();
   172→    throw new ApiError(json.message, res.status, json.code);
   173→  }
   174→  return json.data;
   175→}

   179→/** ورود مدیریت — نام کاربری + رمز → پروفایل ادمین (توکن ذخیره می‌شود) */
   180→export async function adminLogin(username: string, password: string): Promise<AdminProfile> {
   181→  const data = await adminFetch<{ token: string; admin: AdminProfile }>(
   182→    '/admin/auth/login',
   183→    { username, password },
   184→    { auth: false },
   185→  );
   186→  setAdminToken(data.token);
   187→  return data.admin;
   188→}

   203→  try {
   204→    return await adminFetch<AdminProfile>('/admin/auth/me', {});

  1312→/** آپلود پیوست پاسخ پشتیبان — تصویر/ویدئو/سند تا ۵MB */
  1313→export async function uploadAdminSupportFile(file: File): Promise<AdminSupportFileMeta> {
  1314→  const token = getAdminToken();
  1315→  const form = new FormData();
  1316→  form.append('file', file);
  1317→  const res = await fetch(`${apiBase()}/admin/support/uploadFile`, {
  1318→    method: 'POST',
  1319→    headers: token ? { Authorization: `Bearer ${token}` } : {},
  1320→    body: form,
  1321→  });
  1322→  const json = (await res.json().catch(() => null)) as
  1323→    | { status: true; data: AdminSupportFileMeta; message?: string }
  1324→    | { status: false; message: string; code?: string }
  1325→    | null;
  1326→  if (!json) throw new ApiError('آپلود ناموفق بود.', res.status, 'NETWORK');
  1327→  if (json.status === false) {
  1328→    if (res.status === 401) clearAdminToken();
  1329→    throw new ApiError(json.message, res.status, json.code);
  1330→  }
  1331→  return json.data;
  1332→}
